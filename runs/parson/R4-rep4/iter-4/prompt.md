def _encode_json_string(decoded):
    out = ['"']
    for c in decoded:
        if c in _JSON_ESC:
            out.append(_JSON_ESC[c])
        elif ord(c) < 0x20:
            out.append("\\u%04x" % ord(c))
        else:
            out.append(c)
    out.append('"')
    return "".join(out)
